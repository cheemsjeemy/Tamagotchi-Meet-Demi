### 3. Time Signature (src/Demi/Play/Sing.h)

```cpp
struct TimeSignature {
    uint8_t beatsPerMeasure;  // Top number (e.g., 4 in 4/4)
    uint8_t beatUnit;         // Bottom number (e.g., 4 in 4/4)
};

enum NoteDuration {
    WHOLE = 4.0,      // 1/1 (4 beats)
    HALF = 2.0,       // 1/2 (2 beats)
    QUARTER = 1.0,    // 1/4 (1 beat)
    EIGHTH = 0.5,     // 1/8 (1/2 beat)
    SIXTEENTH = 0.25  // 1/16 (1/4 beat)
};
```

### 4. Chord Playback (src/miscellaneous/buzzer.cpp)
Use SigmaDelta to play 3-note chords:

```cpp
void Buzzer::playChord(const String& chordName, float duration, int bpm) {
    auto it = chord_library.find(chordName);
    if (it == chord_library.end()) return;
    
    const Chord& chord = it->second;
    float f1 = midiToFrequency(chord.root);
    float f2 = midiToFrequency(chord.third);
    float f3 = midiToFrequency(chord.fifth);
    
    // Calculate duration in ms
    int durationMs = (int)((duration / bpm) * 60000.0);
    
    // SigmaDelta chord playback
    sigmaDeltaAttach(pin, 312500);
    unsigned long start = millis();
    
    while (millis() - start < durationMs) {
        static float t = 0;
        float wave = (sin(2 * PI * f1 * t) + 
                      sin(2 * PI * f2 * t) + 
                      sin(2 * PI * f3 * t)) / 3.0;
        uint8_t sample = (uint8_t)((wave + 1.0) * 127.5);
        sigmaDeltaWrite(pin, sample);
        t += 0.00005;  // ~20kHz sample rate
        delayMicroseconds(50);
    }
    sigmaDeltaDetach(pin);
}
```

## Usage

```cpp
// Play a chord
buzzer.playChord("C", QUARTER, 120);  // C chord, quarter note, 120 BPM

// Play a single note
buzzer.playNote(60, HALF, 120);  // Middle C, half note, 120 BPM

// Play a song
buzzer.play("ABC Song");  // Existing song playback
```
